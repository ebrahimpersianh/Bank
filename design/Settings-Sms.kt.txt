/*
 * تنظیمات ← پیامکِ بانکی  (زیرصفحه‌ی ۴ از بخشِ ج)
 *
 * کشیده‌نشده. ساخته‌شده از واژگانِ مشترکِ SettingsScreens.kt.txt بخشِ ب.
 * سه نکته‌ی سمتِ کد اعمال شد:
 *   ۱. راهِ «ب» را گرفتم — lastSmsAt. زیرنویسِ همه‌ی ردیف‌ها بر همین بنا شد.
 *   ۲. ردیف «آزمایشِ تشخیص» شد، با پیست‌کردنِ متن. طرحِ کاملش پایین.
 *   ۳. SmsSenderPicker نو نشد؛ جایش در طرح مشخص شده.
 *
 * ══════════════════ چرا این چیدمان ══════════════════
 *
 * این صفحه یک کارِ اصلی دارد: کاربر بفهمد «آیا ثبتِ خودکار دارد کار می‌کند
 * یا نه» — و اگر نه، کدام بانک مشکل دارد. پس ترتیب از تشخیص به عمل می‌رود:
 * کارتِ وضعیت بالا (یک نگاه، جواب می‌گیرد)، بعد فهرستِ بانک‌ها (کدام یکی)،
 * بعد ابزارها (چه کنم).
 *
 * زیرنویسِ هر بانک از lastSmsAt می‌آید و همان حرفی را می‌زند که شمارنده
 * می‌زد، ولی ارزان‌تر. سه حالت دارد و رنگش وضعیت را حمل می‌کند:
 *
 *   سرشماره ثبت نشده        «سرشماره ثبت نشده»        AppLabel
 *   پیامکِ اخیر دارد         «آخرین پیامک: ۳ روز پیش»  AppMuted
 *   بیش از ۳۰ روز بی‌پیامک   «۳۰ روز پیامکی نیامده»    AppDangerInk
 *
 * مرزِ ۳۰ روز را عوض نکن — کوتاه‌ترش کاربرِ کم‌تراکنش را بی‌دلیل می‌ترساند،
 * بلندترش خرابیِ واقعی را دیر می‌گوید. برای حسابی که lastSmsAt هرگز پُر
 * نشده ولی سرشماره دارد، متن «هنوز پیامکی نیامده» با همان AppDangerInk.
 *
 *
 * ══════════════════ تصمیم‌های لازم ══════════════════
 *
 * ۱. کلیدِ اصلیِ صفحه با اجازه‌ی خواندنِ پیامک چه نسبتی دارد؟ اگر کاربر
 *    اجازه را از تنظیماتِ گوشی قطع کرده باشد، کلید روشن است ولی کار نمی‌کند.
 *    پیشنهاد: همان قاعده‌ی ۳۵f — کلید خودکار خاموش نشود، بلکه کارتِ وضعیت
 *    به حالتِ خطا برود و دکمه‌ی «اجازه بده» بگیرد. یعنی کلید حالتِ چهارم
 *    نمی‌گیرد؛ کارتِ بالای صفحه خبر می‌دهد.
 *
 * ۲. حساب‌کتابِ بی‌سرشماره در فهرست بیاید یا نه؟ پیشنهاد: بیاید، ته فهرست،
 *    با کلیدِ خاموش و زیرنویسِ «سرشماره ثبت نشده». چون تنها جایی است که
 *    کاربر می‌فهمد این حساب اصلاً وصل نیست. اگر ترجیح می‌دهی پنهان باشند،
 *    آن‌وقت یک ردیفِ «۲ حسابِ وصل‌نشده» ته فهرست لازم است.
 *
 * ۳. تعدادِ بانک در کارتِ وضعیت: «۳ بانک فعال» یا «۳ از ۵ بانک فعال»؟
 *    پیشنهاد: دومی. کسری کامل‌بودن را می‌گوید و کاربر را به فهرست می‌کشد.
 *
 *
 * ══════════════════ چیدمان ══════════════════
 *
 * سرِ صفحه: دکمه‌ی بازگشتِ ۳۲ + تیترِ ۱۵/۹۰۰ «پیامکِ بانکی».
 * پدینگِ AppSpace.page، فاصله‌ی AppSpace.betweenCards، اسکرول‌شونده،
 * بی‌تبِ پایین.
 *
 * ترتیب:
 *   کارتِ وضعیت
 *   گروهِ «بانک‌ها»   ← یک ردیف به‌ازای هر حساب‌کتاب + ردیفِ افزودن
 *   گروهِ «ابزارها»   ← آزمایشِ تشخیص · قاعده‌های تشخیص
 */

@Composable
fun SmsSettingsScreen(
    masterEnabled: Boolean,
    permissionGranted: Boolean,
    accounts: List<SmsAccount>,
    onMasterToggle: (Boolean) -> Unit,
    onAccountToggle: (SmsAccount, Boolean) -> Unit,
    onPickSender: (SmsAccount) -> Unit,     // SmsSenderPicker را باز می‌کند
    onAddAccount: () -> Unit,
    onTestParser: () -> Unit,
    onRulesClick: () -> Unit,
    onGrantPermission: () -> Unit,
    onBack: () -> Unit,
)

/*
 * ── کارتِ وضعیت ──
 * تنها کارتِ برجسته‌ی صفحه. سه حالت دارد و هر سه یک قالب‌اند:
 * قابِ آیکونِ ۴۰dp گوشه‌ی ۱۲ + متن + کلید یا دکمه. پدینگِ ۱۴، فاصله‌ی ۱۲dp.
 *
 * الف) کار می‌کند — پرِ AppPrimaryPill، حاشیه‌ی ۲dp AppPrimaryBorder،
 *      بی‌سایه. قاب پرِ AppPrimary با آیکونِ ۱۹dp سفید.
 *      عنوان ۱۲٫۵/۹۰۰ AppText: «ثبتِ خودکار روشن است»
 *      زیرنویس ۱۰/۷۰۰ AppPrimaryInk: «۳ از ۵ بانک فعال»
 *      انتها: AppSwitch روشن.
 *
 * ب) خاموش — پرِ AppSurface، حاشیه‌ی ۲dp AppLine.
 *      قاب پرِ AppChipBg با آیکونِ AppLabel.
 *      عنوان همان جا، متنِ «ثبتِ خودکار خاموش است».
 *      زیرنویس ۱۰/۷۰۰ AppMuted: «تراکنش‌ها را دستی وارد می‌کنی»
 *      انتها: AppSwitch خاموش.
 *
 * ج) اجازه ندارد — پرِ AppDangerPill، حاشیه‌ی ۲dp #FFC9C9 (روشن) /
 *      0x57FF6B6B (تیره)، سایه‌ی AppShadow.raised با #FFECEC (روشن).
 *      این تنها حالتی است که سایه می‌گیرد، چون تنها حالتی است که کاربر
 *      باید کاری بکند.
 *      قاب پرِ #FFECEC با آیکونِ AppDanger.
 *      عنوان: «اجازه‌ی خواندنِ پیامک قطع است»
 *      زیرنویس ۱۰/۷۰۰ AppDangerInk با lineHeight ۱٫۷:
 *        «بدونِ این اجازه پیامکِ بانک خوانده نمی‌شود.»
 *      انتها ندارد؛ زیرِ متن یک دکمه‌ی تمام‌عرض با فاصله‌ی ۱۱dp:
 *        پرِ AppPrimary، پدینگِ ۱۲، متنِ ۱۱٫۵/۹۰۰ سفید،
 *        سایه‌ی AppShadow.raised با AppPrimaryDim.
 *      کلیدِ اصلی در این حالت جایش را به دکمه می‌دهد — تصمیمِ ۱ بالا.
 *
 * وقتی کلیدِ اصلی خاموش است، گروهِ بانک‌ها آلفای ۰٫۵ می‌گیرد و کلیک‌ناپذیر
 * می‌شود، ولی پنهان نمی‌شود — کاربر باید ببیند چه چیزی خاموش شده.
 * ردیفِ افزودن و گروهِ ابزارها فعال می‌مانند.
 *
 *
 * ── ردیفِ بانک ──
 * SettingsRow معمولی با سه تفاوت:
 *
 * ۱. جای قابِ آیکون، BankBadge با اندازه‌ی ۳۲ می‌نشیند. اگر BankBadge
 *    اندازه را پارامتر نمی‌گیرد، بگیرد.
 *
 * ۲. عنوان دو تکه است، در یک سطر با فاصله‌ی ۶dp:
 *      نامِ بانک ۱۱٫۵/۸۰۰ AppText
 *      سرشماره ۹/۷۰۰ AppLabel، داخلِ Ltr — ذاتاً چپ‌به‌راست است.
 *    اگر سرشماره نبود، جایش چیپِ «ثبت نشده»: ۸٫۵/۹۰۰ AppLabel،
 *    پرِ AppChipBg، گوشه‌ی ۹۹۹، پدینگِ 3dp×7dp.
 *
 * ۳. انتهای ردیف AppSwitch است. ولی ردیفِ بی‌سرشماره کلید نمی‌گیرد —
 *    چیزی نیست که روشن شود. جایش شِوْرونِ ۱۳dp AppLabel می‌آید و کلیک
 *    SmsSenderPicker را باز می‌کند.
 *    یعنی: سرشماره دارد → کلید. ندارد → شِوْرون. هرگز هر دو.
 *
 * تغییرِ سرشماره‌ی بانکی که از قبل وصل است، از صفحه‌ی خودِ حساب‌کتاب
 * انجام می‌شود، نه اینجا. اینجا فقط روشن و خاموش.
 *
 * ── ردیفِ افزودن ──
 * ته گروه، بعد از جداکننده. قابِ آیکونِ ۳۲ با DashedBorder ۲dp
 * AppDashedBorder و پرِ شفاف، آیکونِ «+» ۱۵dp با AppMuted.
 * متنِ ۱۱٫۵/۸۰۰ AppMuted: «افزودنِ بانک». بی‌زیرنویس، بی‌شِوْرون.
 *
 *
 * ── گروهِ ابزارها ──
 * دو ردیفِ SettingsRow معمولی، هر دو با شِوْرون:
 *
 *   آزمایشِ تشخیص   قابِ سبز    زیرنویس: «متنِ یک پیامک را امتحان کن»
 *   قاعده‌های تشخیص  قابِ خنثی   زیرنویس: «۱۲ قاعده» یا حذف اگر صفر است
 *
 *
 * ══════════════════ صفحه‌ی آزمایشِ تشخیص ══════════════════
 *
 * زیرصفحه‌ی خودش است، نه شیت — چون خروجی‌اش خوانده می‌شود و شیت جا کم دارد.
 * تیترِ ۱۵/۹۰۰ «آزمایشِ تشخیص».
 *
 * بالا: AppTextField چندسطری، ارتفاعِ حداقلِ ۱۰۰dp، گوشه‌ی AppRadius.row،
 * پرِ AppSurface، حاشیه‌ی ۲dp AppLine، پدینگِ ۱۳، متنِ ۱۱/۷۰۰ با
 * lineHeight ۱٫۸. placeholder با AppDisabledText:
 *   «متنِ پیامکِ بانک را اینجا بچسبان»
 * زیرِ فیلد یک سطرِ ۹٫۵/۷۰۰ AppLabel: «متن ذخیره نمی‌شود.»
 *
 * دکمه‌ی «بررسی» تمام‌عرض، AppPrimary، سایه‌ی raised. تا فیلد خالی است
 * حالتِ خاموش می‌گیرد (AppDisabledFill + AppDisabledText، بی‌سایه).
 *
 * ── نتیجه: سه حالت ──
 *
 * الف) شناخته شد — کارتی با پرِ AppPrimaryPill، حاشیه‌ی ۲dp
 *      AppPrimaryBorder، گوشه‌ی card، پدینگِ ۱۴.
 *      سرِ کارت: تیکِ ۱۶dp AppPrimary + «شناسایی شد» ۱۲/۹۰۰.
 *      بعد چهار ردیفِ کلید-مقدار با فاصله‌ی ۹dp و جداکننده‌ی خط‌چینِ ۱dp
 *      به رنگِ AppPrimaryBorder:
 *        مبلغ · نوع · حساب · تاریخ
 *      کلید ۱۰/۷۰۰ AppMuted راست، مقدار ۱۱٫۵/۹۰۰ AppText چپ.
 *      مبلغ رنگ می‌گیرد: برداشت → AppDangerInk، واریز → AppPrimaryInk.
 *      هر فیلدی که موتور درنیاورد، مقدارش «—» با AppDisabledText.
 *
 * ب) شناخته نشد — کارتی با پرِ AppDangerPill، حاشیه‌ی ۲dp #FFC9C9.
 *      عنوان ۱۲/۹۰۰: «شناسایی نشد»
 *      متنِ ۱۰٫۵/۷۰۰ AppDangerInk با lineHeight ۱٫۸، فاصله‌ی ۶dp.
 *      زیرش دو دکمه در یک ردیف با فاصله‌ی ۸dp، هر کدام flex برابر:
 *        «ساختنِ قاعده» — ثانویه (AppSurface + حاشیه‌ی ۲dp AppPrimary
 *                          + متنِ AppPrimaryInk)
 *        «فرستادن برای ما» — خنثی (AppChipBg + متنِ AppMuted)
 *      دومی متن را برای تیم می‌فرستد تا موتور بهتر شود. اگر چنین مسیری
 *      ندارید، حذفش کن و اولی تمام‌عرض شود.
 *
 * ج) پیامکِ بانکی نیست — کارتِ خنثی، پرِ AppSurface2، حاشیه‌ی ۲dp
 *      AppLineRow، بی‌سایه. متنِ ۱۰٫۵/۷۰۰ AppMuted:
 *      «این متن شبیهِ پیامکِ تراکنش نیست.»
 *      بی‌دکمه. این حالت با «ب» فرق دارد: آنجا پیامکِ بانکی بود و پارس
 *      نشد؛ اینجا اصلاً پیامکِ بانکی نبود. اگر موتور این دو را از هم
 *      تشخیص نمی‌دهد، فقط «ب» را پیاده کن و این را ول کن.
 *
 * نتیجه با انیمیشنِ StaggerIn ظاهر شود، از پایین با ۱۲dp جابه‌جایی.
 *
 *
 * ══════════════════ SmsSenderPicker ══════════════════
 *
 * نو نشد. جایش در طرح: شیتِ پایین با گوشه‌ی AppRadius.sheet، از دو مسیر
 * باز می‌شود — شِوْرونِ ردیفِ بی‌سرشماره، و ردیفِ افزودن.
 * دو چیز اگر ندارد، اضافه‌شان ارزش دارد:
 *   دستگیره‌ی ۳۶×۴ با گوشه‌ی ۹۹۹ و رنگِ AppLine در سرِ شیت
 *   چیپِ «قابلِ خواندن» روی فرستنده‌هایی که موتور پارسشان می‌کند:
 *     ۸٫۵/۹۰۰ AppPrimaryInk، پرِ AppPrimaryPill، گوشه‌ی ۹۹۹، پدینگِ 3dp×7dp
 * بقیه‌اش دست نخورد.
 *
 *
 * ══════════════════ مدلِ داده ══════════════════
 *
 * SmsAccount(
 *   id: Long,
 *   bankName: String,
 *   smsSender: String?,        // null یعنی ثبت نشده
 *   enabled: Boolean,
 *   lastSmsAt: Instant?,       // ستونِ تازه — راهِ «ب»
 * )
 *
 * زیرنویس از lastSmsAt ساخته می‌شود، نه از سمتِ UI حساب شود:
 *   smsSender == null            → «سرشماره ثبت نشده» · AppLabel
 *   lastSmsAt == null            → «هنوز پیامکی نیامده» · AppDangerInk
 *   بیش از ۳۰ روز                 → «۳۰ روز پیامکی نیامده» · AppDangerInk
 *   وگرنه                         → «آخرین پیامک: <نسبی>» · AppMuted
 *
 * «<نسبی>» را همان‌جا که بقیه‌ی تاریخ‌های نسبیِ اپ ساخته می‌شوند بساز؛
 * قالبِ تازه لازم نیست. برای امروز «امروز»، دیروز «دیروز»، بعد «n روز پیش».
 *
 * شمارشِ کارتِ وضعیت: فعال = enabled && smsSender != null.
 * مخرج = کلِ حساب‌کتاب‌هایی که در فهرست می‌آیند (تصمیمِ ۲).
 */
