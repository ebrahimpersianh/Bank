@Composable
fun HomeScreen(
    dateLabel: String,              // «سه‌شنبه، ۲۸ مرداد» — از قبل فرمت‌شده
    userName: String,
    streakDays: Int,
    coins: Int,
    today: TodaySpend,
    budget: BudgetSummary,
    dueToday: DueItem?,             // null یعنی کارتِ فوری نمایش داده نمی‌شود
    categories: List<CategoryShare>,
    week: WeekSummary,
    onProfileClick: () -> Unit,
    onBudgetClick: () -> Unit,
    onDuePaid: () -> Unit,
    onWeekClick: () -> Unit,
    onAddClick: () -> Unit,
) {
    val c = /* توکن‌ها را اینجا در val محلی بخوان — Canvas داخلِ دُنات لازمشان دارد */

    Box(Modifier.fillMaxSize().background(AppBg)) {

        Column(Modifier.fillMaxSize()) {

            Column(
                Modifier
                    .weight(1f)
                    .verticalScroll(rememberScrollState())
                    .padding(horizontal = AppSpace.page, vertical = 12.dp),
                verticalArrangement = Arrangement.spacedBy(AppSpace.betweenCards),
            ) {
                GreetingRow(dateLabel, userName, streakDays, coins, onProfileClick)
                TodayHeroCard(today)
                BudgetCard(budget, onBudgetClick)
                dueToday?.let { DueTodayCard(it, onDuePaid) }
                CategoryBreakdownCard(categories)
                WeekReviewCard(week, onWeekClick)
            }

            // تبِ پایین — کامپوننتِ موجود
        }

        AppFab(
            onClick = onAddClick,
            modifier = Modifier
                .align(Alignment.BottomStart)
                .padding(start = AppSpace.page, bottom = 74.dp),
        )
    }
}
